            const dailyTrends = {{ daily_trends|safe }};

                chart: { type: 'line', height: 300, toolbar: { show: true } },
                series: [
                    { name: 'Total Patients', data: totalData },
                    { name: 'New Patients', data: newData },
                    { name: 'Follow-up Patients', data: followUpData }
                ],
                xaxis: { 
                    categories: volumeCategories,
