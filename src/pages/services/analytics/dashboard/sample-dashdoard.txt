{% block js %}
    <script src="{% static 'assets/vendor/apexcharts/apexcharts.min.js' %}"></script>
    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // Patient Volume Trend Chart
            const dailyTrends = {{ daily_trends|safe }};
            const volumeCategories = dailyTrends.map(item => item.day);
            const totalData = dailyTrends.map(item => item.total || 0);
            const newData = dailyTrends.map(item => item.new || 0);
            const followUpData = dailyTrends.map(item => item.follow_up || 0);

            new ApexCharts(document.querySelector("#patient-volume-chart"), {
                chart: { type: 'line', height: 300, toolbar: { show: true } },
                series: [
                    { name: 'Total Patients', data: totalData },
                    { name: 'New Patients', data: newData },
                    { name: 'Follow-up Patients', data: followUpData }
                ],
                xaxis: { 
                    categories: volumeCategories,
                    type: 'datetime',  // Add this for proper date handling
                    labels: {
                        format: 'MMM dd'  // Format for displaying dates
                    }
                },
                colors: ['#3e60d5', '#39afd1', '#47ad77'],
                stroke: { curve: 'smooth', width: 3 },
                markers: { size: 4 }
            }).render();

            // Patient Type Distribution
            const patientTypeData = [
                {{ new_patients|default:0 }},
                {{ follow_up_patients|default:0 }}
            ];

            new ApexCharts(document.querySelector("#patient-type-chart"), {
                chart: { type: 'donut', height: 350 },
                series: patientTypeData,
                labels: ['New Patients', 'Follow-up Patients'],
                colors: ['#39afd1', '#47ad77'],
                legend: { position: 'bottom' }
            }).render();

            // Top Clinics Chart
            const topClinics = {{ top_clinics|safe }};
            const clinicLabels = topClinics.map(item => item.clinic__name);
            const clinicData = topClinics.map(item => item.total_patients);

            new ApexCharts(document.querySelector("#top-clinics-chart"), {
                chart: { type: 'bar', height: 300 },
                series: [{ name: 'Patients', data: clinicData }],
                xaxis: { categories: clinicLabels },
                plotOptions: { 
                    bar: { 
                        horizontal: true, 
                        borderRadius: 4,
                        dataLabels: {
                            position: 'center'
                        }
                    } 
                },
                dataLabels: {
                    enabled: true
                },
                colors: ['#3e60d5']
            }).render();

            // Payment Mode Distribution
            const paymentData = {{ payment_distribution|safe }};
            const paymentLabels = paymentData.map(item => item.payment__name);
            const paymentValues = paymentData.map(item => item.total);

            new ApexCharts(document.querySelector("#payment-mode-chart"), {
                chart: { type: 'pie', height: 350 },
                series: paymentValues,
                labels: paymentLabels,
                legend: { position: 'bottom' },
                dataLabels: {
                    enabled: true,
                    formatter: function (val, opts) {
                        return opts.w.globals.labels[opts.seriesIndex] + ": " + val.toFixed(1) + "%";
                    }
                }
            }).render();
        });
    </script>
{% endblock %}
